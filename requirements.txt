# Audio pipeline + ingestion -- everything except src/video/.
#
#     python -m venv .venv && source .venv/bin/activate
#     pip install -r requirements.txt
#
# Direct dependencies only. pip resolves the rest; this file used to be a raw
# `pip freeze`, which buried the ten packages that actually matter under a
# hundred transitive ones and made it impossible to see what this stage needs.
# To reproduce an exact environment, run `pip freeze > requirements.lock.txt`
# from a working venv and install from that instead.
#
# The face-anonymization stage needs a SEPARATE environment -- see
# requirements-video.txt for why. Do NOT install both into one venv.
#
# The ML stack is pinned because these are the versions the pipeline has
# actually been run against, and torch/whisperx drift breaks alignment quietly
# rather than loudly. The small libraries float.

# src/audio/transcription.py -- ASR, forced alignment, diarization.
# whisperx pulls faster-whisper, ctranslate2 and transformers itself.
whisperx==3.8.6
torch==2.8.0
torchaudio==2.8.0
# Imported as whisperx.diarize.DiarizationPipeline. Pinned rather than left to
# whisperx's resolver because diarization is what separates instructor speech
# from student speech -- if it regresses, students stop being muted.
pyannote-audio==4.0.7

# src/audio/transcription.py, src/audio/captions.py -- question classification
# and caption polish via the Claude API.
anthropic>=0.40
python-dotenv==1.2.2

# src/audio/audio.py -- read and rewrite the muted audio track.
soundfile>=0.12
numpy==2.4.6

# src/audio/cards.py -- render the student-question cards.
pillow==12.3.0

# src/audio/captions.py -- .srt parsing and writing.
pysrt>=1.1

# src/ingestion.py -- browser-driven CMU SSO, then the Panopto API.
playwright==1.61.0
requests>=2.32
